def cargar_imagen_y_mascara(id_tile: str):
    """
    Carga una imagen y su máscara correspondiente.
    
    Args:
        id_tile: Identificador único del tile
        
    Returns:
        imagen_normalizada: Imagen redimensionada y normalizada [0,1]
        m_onehot: Máscara en formato one-hot encoding
    """
    # Construir la ruta de la imagen usando os.path.join en lugar de /
    ruta_imagen = os.path.join(dir_images, f"{id_tile}.tif")
    
    # Verificar si la imagen existe
    if not os.path.exists(ruta_imagen):
        print(f"ID: {id_tile}, Ruta: {ruta_imagen}")
        raise FileNotFoundError(f"No se encuentra la imagen {id_tile}")

    # Cargar imagen original
    imagen = cv2.imread(ruta_imagen)
    if imagen is None:
        raise ValueError(f"No se pudo leer la imagen {ruta_imagen}")
    imagen = cv2.cvtColor(imagen, cv2.COLOR_BGR2RGB)

    # Rasterizar máscara directamente
    alto, ancho = imagen.shape[:2]
    mascara = np.zeros((alto, ancho), dtype=np.uint8)
    for anotacion in anotaciones_por_tile.get(id_tile, []):
        clase = TIPO_A_CLASE.get(anotacion['type'], 0)
        for anillo in anotacion['coordinates']:
            ys = [p[1] for p in anillo]
            xs = [p[0] for p in anillo]
            rr, cc = polygon(ys, xs, shape=mascara.shape)
            mascara[rr, cc] = clase

    # Redimensionar
    imagen_redim = cv2.resize(imagen, TAMANO_IMAGEN)
    mascara_redim = cv2.resize(mascara, TAMANO_IMAGEN, interpolation=cv2.INTER_NEAREST)

    # Codificar máscara en one-hot directamente
    m_onehot = np.zeros(TAMANO_IMAGEN + (NUMERO_CLASES,), dtype=np.uint8)
    for clase_idx in range(NUMERO_CLASES):
        m_onehot[..., clase_idx] = (mascara_redim == clase_idx+1)

    imagen_normalizada = imagen_redim.astype(np.float32) / 255.0
    return imagen_normalizada, m_onehot.astype(np.float32)

def cargar_datos_por_ids(lista_ids, funcion_carga, max_imagenes=MAXIMO_IMAGENES):
    """
    Carga las imágenes y máscaras dado un conjunto de IDs y una función de carga.
    
    Args:
        lista_ids: Lista de identificadores de las imágenes a cargar.
        funcion_carga: Función encargada de cargar cada imagen y máscara.
        max_imagenes: Número máximo de imágenes a cargar.
    
    Returns:
        np.array: Arreglo de imágenes y mascaras cargadas.
    """
    imagenes, mascaras = [], []
    for identificador in lista_ids:
        if len(imagenes) >= max_imagenes:
            break
        try:
            img, msk = funcion_carga(identificador)
            imagenes.append(img)
            mascaras.append(msk)
        except Exception as e:
            print(f"Saltando {identificador}: {e}")
    
    print(f"Se cargaron {len(imagenes)} imágenes.")
    return np.array(imagenes, dtype=np.float32), np.array(mascaras, dtype=np.float32)

def entropia_cruzada_ponderada(pesos_clase: np.ndarray):
    """
    Crea una función de pérdida de entropía cruzada ponderada por clase.
    
    Args:
        pesos_clase: Vector con pesos por clase
        
    Returns:
        función de pérdida para keras
    """
    pesos = tf.constant(pesos_clase, dtype=tf.float32)

    def perdida(y_verdadero: tf.Tensor, y_predicho: tf.Tensor) -> tf.Tensor:
        y_verdadero = tf.cast(y_verdadero, tf.float32)
        y_predicho = tf.clip_by_value(tf.cast(y_predicho, tf.float32), 1e-7, 1.0 - 1e-7)
        perdida_pixel = -tf.reduce_sum(y_verdadero * tf.math.log(y_predicho) * pesos, axis=-1)
        return tf.reduce_mean(perdida_pixel)

    return perdida

def calcular_pesos_clase(mascaras_onehot: np.ndarray) -> np.ndarray:
    """
    Calcula pesos de clase basados en la frecuencia inversa.
    
    Args:
        mascaras_onehot: Array de máscaras en formato one-hot
        
    Returns:
        pesos: Vector con pesos normalizados por clase
    """
    plano = mascaras_onehot.reshape(-1, mascaras_onehot.shape[-1])
    conteos = plano.sum(axis=0)
    total = conteos.sum()
    frecuencia_inversa = total / (conteos + 1e-6)
    pesos = frecuencia_inversa / frecuencia_inversa.sum()
    print(f"Pesos por clase: {pesos}")
    return pesos.astype(np.float32)

def dice_vasos_sanguineos(y_true, y_pred):
    """
    Calcula el coeficiente Dice para la clase de vasos sanguíneos.
    """
    y_true_f = tf.cast(y_true[..., 1], tf.float32)  # Canal 1 = vasos sanguíneos
    y_pred_f = tf.cast(y_pred[..., 1] > 0.5, tf.float32)
    
    interseccion = 2 * tf.reduce_sum(y_true_f * y_pred_f)
    union = tf.reduce_sum(y_true_f + y_pred_f) + 1e-6
    
    return interseccion / union

def iou_promedio(y_verdadero, y_predicho, num_clases):
    """
    Calcula el IoU promedio macro sobre todas las clases.
    """
    ious = []
    for c in range(num_clases):
        verdadero = tf.cast(y_verdadero[..., c], tf.float32)
        predicho = tf.cast(y_predicho[..., c] > 0.5, tf.float32)
        interseccion = tf.reduce_sum(verdadero * predicho)
        union = tf.reduce_sum(verdadero + predicho) - interseccion
        iou = interseccion / (union + 1e-6)
        ious.append(iou)
    return tf.reduce_mean(ious)

def entrenar_con_kfold(
    lista_ids,
    funcion_carga_imagen_mascara,
    funcion_construir_modelo,
    funcion_pesos_clase,
    funcion_loss,
    tamano_imagen,
    num_clases,
    ruta_checkpoints=None,
    n_splits=5,
    epochs=50,
    batch_size=2,
    early_stop_patience=5,
    verbose=1
):
    """
    Entrena un modelo Keras usando validación cruzada K-Fold.
    
    Args:
        lista_ids: Lista de IDs de tiles a utilizar
        funcion_carga_imagen_mascara: Función para cargar imagen y máscara
        funcion_construir_modelo: Función para construir el modelo
        funcion_pesos_clase: Función para calcular pesos de clase
        funcion_loss: Función para construir la función de pérdida
        tamano_imagen: Tamaño de las imágenes (alto, ancho)
        num_clases: Número de clases a segmentar
        ruta_checkpoints: Ruta para guardar checkpoints
        n_splits: Número de folds para cross-validation
        epochs: Número máximo de épocas
        batch_size: Tamaño del batch
        early_stop_patience: Paciencia para early stopping
        verbose: Nivel de verbosidad
        
    Returns:
        historiales: Lista de historiales de entrenamiento por fold
        df_folds: DataFrame con métricas por fold
        df_resumen: DataFrame con resumen de métricas
    """
    ruta_checkpoints = Path(ruta_checkpoints).resolve() if ruta_checkpoints else None
    if ruta_checkpoints:
        ruta_checkpoints.mkdir(parents=True, exist_ok=True)
        (ruta_checkpoints / "historial").mkdir(exist_ok=True)

    kfold = KFold(n_splits=n_splits, shuffle=True, random_state=42)
    historiales = []
    metricas_folds = []

    for fold, (idx_entrenamiento, idx_validacion) in enumerate(tqdm(kfold.split(lista_ids), total=n_splits), start=1):
        print(f"\n--- Fold {fold}/{n_splits} ---")

        ids_entrenamiento = [lista_ids[i] for i in idx_entrenamiento]
        ids_validacion = [lista_ids[i] for i in idx_validacion]

        ruta_pesos = ruta_checkpoints / f"fold_{fold}.weights.h5" if ruta_checkpoints else None
        ruta_historial = ruta_checkpoints / "historial" / f"fold_{fold}_historial.pkl" if ruta_checkpoints else None

        # Limpiar sesión y construir modelo desde cero
        K.clear_session()
        modelo = funcion_construir_modelo()

        # Si ya existe el checkpoint, cargar pesos y saltar entrenamiento
        if ruta_pesos and ruta_pesos.exists():
            print("Checkpoint encontrado, cargando pesos y omitiendo entrenamiento.")
            modelo.load_weights(str(ruta_pesos))
            pesos_dummy = np.ones(num_clases, dtype=np.float32)
            modelo.compile(
                optimizer=Adam(1e-3),
                loss=funcion_loss(pesos_dummy),
                metrics=[dice_vasos_sanguineos, lambda y_true, y_pred: iou_promedio(y_true, y_pred, num_clases)]
            )
            if ruta_historial and ruta_historial.exists():
                with open(ruta_historial, "rb") as f:
                    historial = pickle.load(f)
                historiales.append(historial)
            else:
                historiales.append({})
        else:
            # Cargar datos de entrenamiento
            X_train, Y_train = cargar_datos_por_ids(ids_entrenamiento, funcion_carga_imagen_mascara)
            pesos_clase = funcion_pesos_clase(Y_train)
            
            # Compilar modelo
            modelo.compile(
                optimizer=Adam(1e-5),
                loss=funcion_loss(pesos_clase),
                metrics=[dice_vasos_sanguineos, lambda y_true, y_pred: iou_promedio(y_true, y_pred, num_clases)]
            )

            # Callbacks para control de entrenamiento
            callbacks = [
                EarlyStopping(
                    monitor='dice_vasos_sanguineos', 
                    patience=early_stop_patience, 
                    mode='max',
                    restore_best_weights=True, 
                    verbose=1
                ),
            ]
            
            if ruta_pesos:
                callbacks.append(ModelCheckpoint(
                    filepath=str(ruta_pesos),
                    monitor='dice_vasos_sanguineos',
                    mode='max',
                    save_best_only=True,
                    save_weights_only=True,
                    verbose=1
                ))

            # Entrenamiento del modelo
            X_val, Y_val = cargar_datos_por_ids(ids_validacion, funcion_carga_imagen_mascara)
            
            historial_obj = modelo.fit(
                X_train, Y_train,
                batch_size=batch_size,
                validation_data=(X_val, Y_val),
                epochs=epochs,
                verbose=verbose,
                callbacks=callbacks
            )
            
            # Guardar historial del fold
            if ruta_historial:
                with open(ruta_historial, "wb") as f:
                    pickle.dump(historial_obj.history, f)
            
            historiales.append(historial_obj.history)
            
            # Recargar los mejores pesos
            if ruta_pesos and ruta_pesos.exists():
                modelo.load_weights(str(ruta_pesos))

        # Evaluación en conjunto de validación
        X_val, Y_val = cargar_datos_por_ids(ids_validacion, funcion_carga_imagen_mascara)
        probabilidades = modelo.predict(X_val, verbose=0)  # (B,H,W,C)
        predicciones = np.argmax(probabilidades, axis=-1).ravel()
        verdaderos = np.argmax(Y_val, axis=-1).ravel()

        # Métricas para vasos sanguíneos (clase 1)
        interseccion = np.sum((verdaderos == 1) & (predicciones == 1))
        dice = 2 * interseccion / (np.sum(verdaderos == 1) + np.sum(predicciones == 1) + 1e-6)
        union = np.sum((verdaderos == 1) | (predicciones == 1))
        iou_vasos = interseccion / (union + 1e-6)

        # Métricas multiclase
        precision = precision_score(verdaderos, predicciones, average='macro', zero_division=0)
        recall = recall_score(verdaderos, predicciones, average='macro', zero_division=0)
        bal_acc = balanced_accuracy_score(verdaderos, predicciones)

        # AUC OvO macro
        y_true_oh = np.eye(num_clases)[verdaderos]
        probs_flat = probabilidades.reshape(-1, num_clases)
        try:
            auc = roc_auc_score(y_true_oh, probs_flat, multi_class='ovo', average='macro')
        except:
            auc = np.nan

        metricas_folds.append({
            'fold': fold,
            'Dice_vasos': dice,
            'IoU_vasos': iou_vasos,
            'Precision': precision,
            'Recall': recall,
            'AUC_macro': auc,
            'ExactitudBalanceada': bal_acc
        })

    df_folds = pd.DataFrame(metricas_folds)
    df_resumen = pd.DataFrame({
        'media': df_folds.mean(numeric_only=True),
    }).T

    return historiales, df_folds, df_resumen

def graficar_curvas_entrenamiento(historiales, metrica='loss', val_metrica='val_loss'):

def graficar_pr_roc(modelo, ids, funcion_cargar, num_clases):
    """
    Grafica curvas PR y ROC promediadas (micro-average).
    
    Args:
        modelo: Modelo Keras entrenado
        ids: Lista de IDs para evaluar
        funcion_cargar: Función para cargar imagen y máscara
        num_clases: Número de clases
    """
    X, Y = cargar_datos_por_ids(ids, funcion_cargar)
    
    probs = modelo.predict(X, verbose=0)
    probs_flat = probs.reshape(-1, num_clases)
    
    y_flat = np.argmax(Y, axis=-1).ravel()
    y_true_oh = np.eye(num_clases, dtype=int)[y_flat]
    
    y_micro = y_true_oh.ravel()
    p_micro = probs_flat.ravel()
    
    # Curva PR
    prec, rec, _ = precision_recall_curve(y_micro, p_micro)
    plt.figure(figsize=(6, 5))
    plt.plot(rec, prec, label='Curva PR (micro-avg)')
    plt.hlines(y_micro.mean(), 0, 1, linestyles='--', label=f'Línea base={y_micro.mean():.3f}')
    plt.title('Precisión-Exhaustividad Micro-promediada')
    plt.xlabel('Exhaustividad (Recall)')
    plt.ylabel('Precisión')
    plt.legend()
    plt.grid(True)
    plt.show()
    
    # Curva ROC
    fpr, tpr, _ = roc_curve(y_micro, p_micro)
    auc_m = sklearn_auc(fpr, tpr)
    plt.figure(figsize=(6, 5))
    plt.plot(fpr, tpr, label=f'ROC (micro-avg) AUC={auc_m:.3f}')
    plt.plot([0, 1], [0, 1], linestyle='--', label='Aleatorio')
    plt.title('ROC Micro-promediada')
    plt.xlabel('Tasa de Falsos Positivos')
    plt.ylabel('Tasa de Verdaderos Positivos')
    plt.legend()
    plt.grid(True)
    plt.show()

def graficar_mapas_error(modelo, ids, funcion_cargar, num_clases, n_muestras=3):
    """
    Muestra mapas de error para n_muestras aleatorias.
    
    Args:

        ids: Lista de IDs para evaluar

        pred = modelo.predict(img[None], verbose=0)[0]
        pc = np.argmax(pred, axis=-1)
        tc = np.argmax(y_true, axis=-1)
        
        # Falsos positivos y falsos negativos para vasos sanguíneos (clase 1)
        fp = (pc == 1) & (tc != 1)
        fn = (pc != 1) & (tc == 1)
        
        # Crear superposición de errores
        overlay = img.copy()
        overlay[fp] = [1, 0, 0]  # rojo = FP
        overlay[fn] = [0, 0, 1]  # azul = FN
        
        # Graficar los 4 paneles
        plt.figure(figsize=(16, 4))
        
        plt.subplot(1, 4, 1)
        plt.imshow(img)
        plt.title('Imagen')
        plt.axis('off')
        
        plt.subplot(1, 4, 2)
        plt.imshow(tc, cmap='gray', vmin=0, vmax=num_clases-1)
        plt.title('Máscara Real')
        plt.axis('off')
        
        plt.subplot(1, 4, 3)
        plt.imshow(pc, cmap='gray', vmin=0, vmax=num_clases-1)
        plt.title('Predicción')
        plt.axis('off')
        
        plt.subplot(1, 4, 4)
        plt.imshow(overlay)
        plt.title('Errores (FP rojo, FN azul)')
        plt.axis('off')
        
        plt.tight_layout()
        plt.show()

# Ejemplo de entrenamiento y evaluación completa
def ejecutar_pipeline_completo(
    lista_ids,
    funcion_cargar,
    funcion_construir_modelo,
    ruta_base,
    n_splits=5,
    epochs=50,
    batch_size=2
):
    """
    Ejecuta el pipeline completo: entrenamiento, evaluación y visualización.
    
    Args:
        lista_ids: Lista de IDs de tiles
        funcion_cargar: Función para cargar imagen y máscara
        funcion_construir_modelo: Función para construir el modelo
        ruta_base: Ruta base para guardar checkpoints
        n_splits: Número de folds
        epochs: Número máximo de épocas
        batch_size: Tamaño del batch
    
    Returns:
        mejor_modelo: El mejor modelo entrenado
        df_resumen: DataFrame con resumen de métricas
    """
    directorio_checkpoints = crear_directorio_checkpoints(ruta_base)
    
    historiales, df_folds, df_resumen = entrenar_con_kfold(
        lista_ids=lista_ids,
        funcion_carga_imagen_mascara=funcion_cargar,
        funcion_construir_modelo=funcion_construir_modelo,
        funcion_pesos_clase=calcular_pesos_clase,
        funcion_loss=entropia_cruzada_ponderada,
        tamano_imagen=TAMANO_IMAGEN,
        num_clases=NUMERO_CLASES,
        ruta_checkpoints=directorio_checkpoints,
        n_splits=n_splits,
        epochs=epochs,
        batch_size=batch_size,
        early_stop_patience=10,
        verbose=1
    )
    
    print("\n--- Resumen de Métricas ---")
    print(df_resumen)
    
    # Visualizaciones
    graficar_curvas_entrenamiento(historiales, 'loss', 'val_loss')
    graficar_curvas_entrenamiento(historiales, 'dice_vasos_sanguineos', 'val_dice_vasos_sanguineos')
    
    graficar_boxplots_metricas(df_folds)
    graficar_histogramas_metricas(df_folds)
    
    # Seleccionar el mejor modelo

def modelo_unet(input_shape, num_classes):
    inputs = layers.Input(shape=input_shape)
    
    def conv_block(x, filters):
        x = layers.Conv2D(filters, 3, padding='same')(x)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        x = layers.Conv2D(filters, 3, padding='same')(x)
        x = layers.BatchNormalization()(x)
        x = layers.ReLU()(x)
        return x
    
    def encoder_block(x, filters):
        x = conv_block(x, filters)
        p = layers.MaxPooling2D((2, 2))(x)
        return x, p
    
    def decoder_block(x, skip, filters):
        x = layers.Conv2DTranspose(filters, 2, strides=2, padding='same')(x)
        x = layers.Concatenate()([x, skip])
        return conv_block(x, filters)
    
    f1, p1 = encoder_block(inputs, 32)
    f2, p2 = encoder_block(p1, 64)
    f3, p3 = encoder_block(p2, 128)
    
    b = conv_block(p3, 256)
    
    d3 = decoder_block(b, f3, 128)
    d2 = decoder_block(d3, f2, 64)
    d1 = decoder_block(d2, f1, 32)
    
    outputs = layers.Conv2D(num_classes, 1, activation='softmax')(d1)
    
    return models.Model(inputs, outputs)

from tensorflow.keras import layers, models
from tensorflow.keras.layers import LeakyReLU

# Bloque de convolución con normalización y activación
def bloque_convolucional(x, filtros):
    x = layers.Conv2D(filtros, 3, padding='same', kernel_initializer='he_normal')(x)
    x = layers.BatchNormalization(momentum=0.9)(x)
    return LeakyReLU(alpha=0.1)(x)

# Bloque de downsampling con conexión residual
def bloque_downsampling(x, filtros):
    x = bloque_convolucional(x, filtros)
    x = bloque_convolucional(x, filtros)
    conexion_residual = x
    x = layers.MaxPooling2D(pool_size=(2, 2))(x)
    return x, conexion_residual

# Bloque de upsampling con conexión residual
def bloque_upsampling(x, conexion_residual, filtros):
    x = layers.Conv2DTranspose(filtros, 2, strides=2, padding='same')(x)
    x = layers.Concatenate()([x, conexion_residual])
    return bloque_convolucional(bloque_convolucional(x, filtros), filtros)

# Construcción del modelo V-Net
def construir_vnet(tamano_entrada, num_clases):
    entradas = layers.Input(shape=tamano_entrada)

    # Encoder
    x1, conexion1 = bloque_downsampling(entradas, 32)   
    x2, conexion2 = bloque_downsampling(x1, 64)       
    x3, conexion3 = bloque_downsampling(x2, 128)      
    x4, conexion4 = bloque_downsampling(x3, 256)      

    # Bottleneck
    x = bloque_convolucional(x4, 512)
    x = bloque_convolucional(x, 512)

    # Decoder
    x = bloque_upsampling(x, conexion4, 256)         
    x = bloque_upsampling(x, conexion3, 128)         
    x = bloque_upsampling(x, conexion2, 64)

# Bloque de convolución con dropout y normalización
def bloque_convolucional(x, filtros, kernel=3, dropout=0.1, bn_momentum=0.9):
    x = layers.Conv2D(filtros, kernel, padding='same', kernel_initializer='he_normal')(x)
    x = layers.BatchNormalization(momentum=bn_momentum)(x)
    x = LeakyReLU(alpha=0.1)(x)
    x = SpatialDropout2D(dropout)(x)
    x = layers.Conv2D(filtros, kernel, padding='same', kernel_initializer='he_normal')(x)
    x = layers.BatchNormalization(momentum=bn_momentum)(x)
    x = LeakyReLU(alpha=0.1)(x)
    x = SpatialDropout2D(dropout)(x)
    return x

# Bloque de encoder con max pooling
def bloque_encoder(x, filtros):
    x = bloque_convolucional(x, filtros)
    x = layers.MaxPooling2D((2, 2))(x)
    return x

# Bloque de decoder con transposed convolution y concatenación
def bloque_decoder(x, skip, filtros):
    x = layers.Conv2DTranspose(filtros, 2, strides=2, padding='same')(x)
    x = layers.Concatenate()([x, skip])
    x = bloque_convolucional(x, filtros)
    return x

# Modelo SegNet simplificado
def construir_segnet(entrada, num_clases):
    x_input = layers.Input(shape=entrada)

    # Encoder con almacenamiento de skips antes de pooling
    c1 = bloque_convolucional(x_input, 64)
    e1 = layers.MaxPooling2D((2, 2))(c1)

    c2 = bloque_convolucional(e1, 128)
    e2 = layers.MaxPooling2D((2, 2))(c2)

    c3 = bloque_convolucional(e2, 256)
    e3 = layers.MaxPooling2D((2, 2))(c3)

    c4 = bloque_convolucional(e3, 512)
    e4 = layers.MaxPooling2D((2, 2))(c4)

    # Bottleneck
    b = bloque_convolucional(e4, 512)

    # Decoder (uso de skips correctos del encoder)
    d4 = bloque_decoder(b, c4, 512)
    d3 = bloque_decoder(d4, c3, 256)
    d2 = bloque_decoder(d3, c2, 128)
    d1 = bloque_decoder(d2, c1, 64)

    # Salida
    salidas = layers.Conv2D(num_clases, 1, activation='softmax')(d1)

    return models.Model(x_input, salidas)

def bloque_aspp(x, filtros):
    y1 = layers.Conv2D(filtros, 1, padding='same', dilation_rate=1, activation='relu')(x)
    y2 = layers.Conv2D(filtros, 3, padding='same', dilation_rate=6, activation='relu')(x)
    y3 = layers.Conv2D(filtros, 3, padding='same', dilation_rate=12, activation='relu')(x)
    y4 = layers.Conv2D(filtros, 3, padding='same', dilation_rate=18, activation='relu')(x)

    h, w = x.shape[1], x.shape[2]
    y5 = layers.GlobalAveragePooling2D()(x)
    y5 = layers.Reshape((1, 1, x.shape[-1]))(y5)
    y5 = layers.UpSampling2D(size=(h, w), interpolation='bilinear')(y5)

    y = layers.Concatenate()([y1, y2, y3, y4, y5])
    return layers.Conv2D(filtros, 1, padding='same', activation='relu')(y)

from tensorflow.keras import layers, models

def puerta_atencion(skip_conn, gating, canales_intermedios):
    θ_x = layers.Conv2D(canales_intermedios, 1, padding='same')(skip_conn)
    φ_g = layers.Conv2D(canales_intermedios, 1, padding='same')(gating)
    suma = layers.Add()([θ_x, φ_g])
    relu = layers.Activation('relu')(suma)
    ψ = layers.Conv2D(1, 1, padding='same')(relu)
    sigmoide = layers.Activation('sigmoid')(ψ)
    return layers.Multiply()([skip_conn, sigmoide])

def bloque_conv(x, filtros, tamano_kernel=3, momentum_bn=0.9, tasa_dropout=0.1):
    x = layers.Conv2D(filtros, tamano_kernel, padding='same', kernel_initializer='he_normal')(x)
    x = layers.BatchNormalization(momentum=momentum_bn)(x)
    x = layers.LeakyReLU(alpha=0.1)(x)
    x = layers.SpatialDropout2D(tasa_dropout)(x)
    x = layers.Conv2D(filtros, tamano_kernel, padding='same', kernel_initializer='he_normal')(x)
    x = layers.BatchNormalization(momentum=momentum_bn)(x)
    x = layers.LeakyReLU(alpha=0.1)(x)
    x = layers.SpatialDropout2D(tasa_dropout)(x)
    return x

def bloque_encoder(x, filtros):
    feats = bloque_conv(x, filtros)
    pooled = layers.MaxPooling2D((2,2))(feats)
    return feats, pooled

def bloque_decoder(x, skip_feats, filtros):
    up = layers.UpSampling2D((2,2), interpolation='bilinear')(x)
    attn = puerta_atencion(skip_feats, up, canales_intermedios=filtros//2)
    merged = layers.Concatenate()([up, attn])
    return bloque_conv(merged, filtros)

def construir_unet_con_atencion(tamano_entrada=(128,128,3), num_clases=2, filtros_base=32):
    inputs = layers.Input(shape=tamano_entrada)
    # Encoder
    f1, p1 = bloque_encoder(inputs, filtros_base)
    f2, p2 = bloque_encoder(p1, filtros_base*2)
    f3, p3 = bloque_encoder(p2, filtros_base*4)
    f4, p4 = bloque_encoder(p3, filtros_base*8)
    # Bottleneck
    b, _ = bloque_encoder(p4, filtros_base*16)
    # Decoder
    d4 = bloque_decoder(b, f4, filtros_base*8)
    d3 = bloque_decoder(d4, f3, filtros_base*4)
    d2 = bloque_decoder(d3, f2, filtros_base*2)
    d1 = bloque_decoder(d2, f1, filtros_base)
    # Output
    outputs = layers.Conv2D(num_clases, 1, activation='softmax')(d1)
    return models.Model(inputs, outputs)

from tensorflow.keras import layers, models

# Función Conv2D + BatchNormalization + ReLU
def conv_bn_relu(x, filtros, tamano_kernel, padding='same', nombre=None):
    x = layers.Conv2D(filtros, tamano_kernel, padding=padding, kernel_initializer='he_normal', name=f"{nombre}_conv" if nombre else None)(x)
    x = layers.BatchNormalization(name=f"{nombre}_bn" if nombre else None)(x)
    return layers.Activation('relu', name=f"{nombre}_act" if nombre else None)(x)

# Bloque residual
def bloque_residual(x, filtros, nombre=None):
    atajo = layers.Conv2D(filtros, 1, padding='same', kernel_initializer='he_normal', name=f"{nombre}_shortcut_conv")(x)
    atajo = layers.BatchNormalization(name=f"{nombre}_shortcut_bn")(atajo)

    x = layers.Conv2D(filtros, 3, padding='same', kernel_initializer='he_normal', name=f"{nombre}_conv1")(x)
    x = layers.BatchNormalization(name=f"{nombre}_bn1")(x)
    x = layers.Activation('relu', name=f"{nombre}_act1")(x)

    x = layers.Conv2D(filtros, 3, padding='same', kernel_initializer='he_normal', name=f"{nombre}_conv2")(x)
    x = layers.BatchNormalization(name=f"{nombre}_bn2")(x)

    return layers.Activation('relu', name=f"{nombre}_out")(layers.Add()([atajo, x]))

# Construir ResUNet
def construir_resunet(tamano_entrada=(256, 256, 3), num_clases=1, filtros_base=64):
    inputs = layers.Input(shape=tamano_entrada, name="input")

    # Encoder
    c1 = conv_bn_relu(inputs, filtros_base, 3, nombre="enc1")
    p1 = layers.MaxPooling2D(name="pool1")(c1)

    c2 = bloque_residual(p1, filtros_base*2, nombre="enc2")
    p2 = layers.MaxPooling2D(name="pool2")(c2)

    c3 = bloque_residual(p2, filtros_base*4, nombre="enc3")
    p3 = layers.MaxPooling2D(name="pool3")(c3)

    c4 = bloque_residual(p3, filtros_base*8, nombre="enc4")
    p4 = layers.MaxPooling2D(name="pool4")(c4)

    # Bridge
    b = bloque_residual(p4, filtros_base*16, nombre="bridge")

    # Decoder
    u4 = layers.UpSampling2D(name="up4")(b)
    u4 = layers.Concatenate(name="concat4")([u4, c4])
    d4 = conv_bn_relu(u4, filtros_base*8, 3, nombre="dec4")

    u3 = layers.UpSampling2D(name="up3")(d4)
    u3 = layers.Concatenate(name="concat3")([u3, c3])
    d3 = conv_bn_relu(u3, filtros_base*4, 3, nombre="dec3")

    u2 = layers.UpSampling2D(name="up2")(d3)
    u2 = layers.Concatenate(name="concat2")([u2, c2])
    d2 = conv_bn_relu(u2, filtros_base*2, 3, nombre="dec2")

    u1 = layers.UpSampling2D(name="up1")(d2)
    u1 = layers.Concatenate(name="concat1")([u1, c1])
    d1 = conv_bn_relu(u1, filtros_base, 3, nombre="dec1")

    # Salida
    outputs = layers.Conv2D(num_clases, 1, activation='sigmoid', name="output")(d1)

    return models.Model(inputs=inputs, outputs=outputs, name="ResUNet")

from tensorflow.keras import layers, models

# Función auxiliar Conv2D + BatchNormalization + ReLU
def conv_bn_relu(x, filtros, tamano_kernel, padding='same', nombre=None):
    x = layers.Conv2D(filtros, tamano_kernel, padding=padding, name=f"{nombre}_conv" if nombre else None)(x)
    x = layers.BatchNormalization(name=f"{nombre}_bn" if nombre else None)(x)
    return layers.Activation('relu', name=f"{nombre}_act" if nombre else None)(x)

# Constructor del modelo PSPNet
def construir_pspnet(tamano_entrada, num_clases):
    entradas = layers.Input(shape=tamano_entrada)
    
    # Codificador: bloques iniciales de convolución
    x = conv_bn_relu(entradas, 64, 3, nombre='bloque_codificador1')
    x = conv_bn_relu(x, 64, 3, nombre='bloque_codificador2')
    
    # Módulo de Agrupamiento Piramidal (PPM)
    tamanos_pooling = [1, 2, 4, 8]
    salidas_pooling = [
        layers.UpSampling2D(size=(tamano, tamano), interpolation='bilinear', name=f'ppm_upsample_{tamano}')(
            conv_bn_relu(layers.AveragePooling2D(pool_size=(tamano, tamano), name=f'ppm_pool_{tamano}')(x), 32, 1, nombre=f'ppm_conv_{tamano}')
        ) for tamano in tamanos_pooling
    ]
    
    # Concatenar los niveles de la pirámide
    x = layers.Concatenate(name='ppm_concat')([x] + salidas_pooling)
    
    # Decodificador: bloque final de convolución
    x = conv_bn_relu(x, 64, 3, nombre='bloque_decodificador1')
    
    # Capa de salida
    salidas = layers.Conv2D(num_clases, 1, activation='softmax', name='salida_pspnet')(x)
    
    return models.Model(inputs=entradas, outputs=salidas, name='PSPNet')

from tensorflow.keras import layers, models

# Función auxiliar: Conv2D + BatchNormalization + ReLU
def conv_bn_relu(x, filtros, tamano_kernel, padding='same'):
    """Aplica Conv2D + BatchNormalization + ReLU."""
    # Capa Conv2D
    x = layers.Conv2D(filtros, tamano_kernel, padding=padding, kernel_initializer='he_normal')(x)
    # Capa de normalización
    x = layers.BatchNormalization()(x)
    # Capa de activación ReLU
    return layers.Activation('relu')(x)

# Modelo FCN-8s
def construir_fcn(tamano_entrada, num_clases=3, filtros_base=42):
    """Construye el modelo FCN-8s para segmentación de imágenes."""
    
    # Entrada
    entradas = layers.Input(shape=tamano_entrada)
    x = entradas
    saltos = []  # Guardará las salidas de las capas de salto

    # Codificador (Encoder) con bloques convolucionales y max pooling
    for idx_bloque, multiplicador in enumerate([1, 2, 4, 8, 8], start=1):
        filtros = filtros_base * multiplicador
        for _ in range(2 if idx_bloque < 3 else 3):
            x = conv_bn_relu(x, filtros, 3)  # Conv2D + BatchNorm + ReLU
        pool = layers.MaxPooling2D(pool_size=(2, 2))(x)  # MaxPooling
        saltos.append(x)  # Guardamos las salidas antes del pooling
        x = pool  # Actualizamos 'x' con el valor después del pooling

    # Capas completamente convolucionales (Fully Convolutional)
    x = layers.Conv2D(filtros_base * 32, 7, padding='same', activation='relu')(x)
    x = layers.Dropout(0.5)(x)  # Dropout para evitar sobreajuste
    x = layers.Conv2D(filtros_base * 32, 1, padding='same', activation='relu')(x)
    x = layers.Dropout(0.5)(x)

    # Capas de puntuación (score layers) utilizando las salidas de las capas de salto
    score_pool3 = layers.Conv2D(num_clases, 1, padding='same')(saltos[3])  # Salida de pool3
    score_pool4 = layers.Conv2D(num_clases, 1, padding='same')(saltos[4])  # Salida de pool4
    score_fc7   = layers.Conv2D(num_clases, 1, padding='same')(x)  # Salida de las capas completamente convolucionales

    # Upsampling y fusión de las características para lograr la segmentación final
    upscore2 = layers.Conv2DTranspose(num_clases, 4, strides=2, padding='same')(score_fc7)  # 8->16
    fuse_pool4 = layers.Concatenate()([upscore2, score_pool4])  # Fusión de la capa upscore2 con score_pool4

    upscore4 = layers.Conv2DTranspose(num_clases, 4, strides=2, padding='same')(fuse_pool4)  # 16->32
    fuse_pool3 = layers.Concatenate()([upscore4, score_pool3])  # Fusión de la capa upscore4 con score_pool3

    # Upsampling final
    upscore8 = layers.Conv2DTranspose(num_clases, 16, strides=8, padding='same')(fuse_pool3)  # 32->256

    # Capa de salida (softmax para clasificación multiclase)
    salidas = layers.Conv2D(num_clases, 1, activation='softmax', padding='same')(upscore8)

    # Definición del modelo
    modelo = models.Model(inputs=entradas, outputs=salidas)
    return modelo
